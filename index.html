const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS, GET", "Access-Control-Allow-Headers": "Content-Type", }; export default { async fetch(request, env) { // ========================= // CORS // ========================= if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } // ========================= // TEST WORKER // ========================= if (request.method === "GET") { return new Response( JSON.stringify({ ok: true, message: "ChatLuk AI Worker đang hoạt động 🐉" }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // ========================= // CHỈ NHẬN POST // ========================= if (request.method !== "POST") { return new Response( JSON.stringify({ error: "Method not allowed" }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } try { // ========================= // KIỂM TRA API KEY // ========================= if (!env.OPENAI_API_KEY) { return new Response( JSON.stringify({ error: "OPENAI_API_KEY chưa được cấu hình trong Worker." }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // ========================= // ĐỌC REQUEST // ========================= const body = await request.json(); const message = body.message; if ( typeof message !== "string" || !message.trim() ) { return new Response( JSON.stringify({ error: "Tin nhắn không hợp lệ." }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // ========================= // GỌI OPENAI // ========================= const openaiResponse = await fetch( "https://api.openai.com/v1/responses", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${env.OPENAI_API_KEY}` }, body: JSON.stringify({ model: "gpt-5-mini", instructions: "Bạn là ChatLuk, một trợ lý AI nói tiếng Việt. " + "Hãy trả lời tự nhiên, hữu ích, rõ ràng và thân thiện. " + "Có thể sử dụng emoji vừa phải. " + "Không tự nhận mình là ChatGPT.", input: message }) } ); // ========================= // ĐỌC RESPONSE OPENAI // ========================= const responseText = await openaiResponse.text(); let data; try { data = JSON.parse(responseText); } catch { return new Response( JSON.stringify({ error: "OpenAI trả về dữ liệu không hợp lệ.", details: responseText.substring(0, 500) }), { status: 502, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // ========================= // OPENAI ERROR // ========================= if (!openaiResponse.ok) { return new Response( JSON.stringify({ error: data?.error?.message || "OpenAI API request failed.", code: data?.error?.code || null }), { status: openaiResponse.status, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // ========================= // LẤY TEXT // ========================= let answer = ""; if (typeof data.output_text === "string") { answer = data.output_text; } // fallback nếu output_text không có if (!answer && Array.isArray(data.output)) { for (const item of data.output) { if ( item.type === "message" && Array.isArray(item.content) ) { for (const content of item.content) { if ( content.type === "output_text" && typeof content.text === "string" ) { answer += content.text; } } } } } if (!answer) { answer = "ChatLuk chưa nhận được nội dung trả lời từ AI."; } // ========================= // TRẢ KẾT QUẢ VỀ WEBSITE // ========================= return new Response( JSON.stringify({ ok: true, reply: answer }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } catch (error) { console.error(error); return new Response( JSON.stringify({ error: error?.message || "Worker gặp lỗi không xác định." }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } } };
