const MODEL = "gemini-3.6-flash"; const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json; charset=UTF-8" }; function responseJSON(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: CORS_HEADERS }); } export default { async fetch(request, env) { // ============================== // CORS // ============================== if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: CORS_HEADERS }); } const url = new URL(request.url); // ============================== // KIỂM TRA WORKER // ============================== if ( request.method === "GET" && url.pathname === "/" ) { return responseJSON({ ok: true, service: "ChatLuk AI", message: "ChatLuk Worker đang hoạt động.", model: MODEL }); } // ============================== // API CHAT // ============================== if ( request.method === "POST" && url.pathname === "/api/chat" ) { try { if (!env.GEMINI_API_KEY) { return responseJSON({ ok: false, error: "Chưa cấu hình GEMINI_API_KEY." }, 500); } const requestData = await request.json(); const message = typeof requestData.message === "string" ? requestData.message.trim() : ""; if (!message) { return responseJSON({ ok: false, error: "Tin nhắn trống." }, 400); } // ============================== // GỌI GEMINI // ============================== const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent`; const geminiResponse = await fetch( endpoint, { method: "POST", headers: { "Content-Type": "application/json", "x-goog-api-key": env.GEMINI_API_KEY }, body: JSON.stringify({ contents: [ { role: "user", parts: [ { text: `Bạn là ChatLuk, một trợ lý AI Việt Nam. Hãy trả lời bằng tiếng Việt tự nhiên, thân thiện, dễ hiểu và hữu ích. Không tự nhận mình là ChatGPT. Không nói về API, Cloudflare hoặc mã nguồn nếu người dùng không hỏi. Tin nhắn người dùng: ${message}` } ] } ] }) } ); // ============================== // ĐỌC KẾT QUẢ GEMINI // ============================== const raw = await geminiResponse.text(); let data; try { data = JSON.parse(raw); } catch (error) { console.error("Gemini trả về không phải JSON:", raw); return responseJSON({ ok: false, error: "Gemini không trả về JSON hợp lệ.", details: raw.substring(0, 1000) }, 502); } // ============================== // GEMINI BÁO LỖI // ============================== if (!geminiResponse.ok) { console.error("Gemini API ERROR:", data); return responseJSON({ ok: false, error: data?.error?.message || "Gemini API trả về lỗi.", status: geminiResponse.status }, geminiResponse.status); } // ============================== // LẤY CÂU TRẢ LỜI // ============================== const answer = data?.candidates?.[0]?.content?.parts ?.map(part => part.text || "") .join("") .trim(); if (!answer) { console.error("Gemini không có nội dung:", data); return responseJSON({ ok: false, error: "Gemini không trả về nội dung." }, 502); } // ============================== // TRẢ CHO CHATLUK // ============================== return responseJSON({ ok: true, answer }); } catch (error) { console.error("CHATLUK WORKER ERROR:", error); return responseJSON({ ok: false, error: error?.message || "Worker xảy ra lỗi không xác định." }, 500); } } return responseJSON({ ok: false, error: "ChatLuk API không tồn tại." }, 404); } };
