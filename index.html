const MODEL = "gemini-3.6-flash"; const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json; charset=UTF-8" }; function responseJSON(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: CORS_HEADERS }); } export default { async fetch(request, env) { // ============================== // CORS // ============================== if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: CORS_HEADERS }); } const url = new URL(request.url); // ============================== // KIỂM TRA WORKER // ============================== if ( request.method === "GET" && url.pathname === "/" ) { return responseJSON({ ok: true, service: "ChatLuk AI", message: "ChatLuk Worker đang hoạt động.", model: MODEL }); } // ============================== // API CHAT // ============================== if ( request.method === "POST" && url.pathname === "/api/chat" ) { try { // Kiểm tra API key if (!env.GEMINI_API_KEY) { return responseJSON({ ok: false, error: "Không tìm thấy GEMINI_API_KEY trong Cloudflare Worker." }, 500); } // Đọc request const requestData = await request.json(); const message = typeof requestData.message === "string" ? requestData.message.trim() : ""; if (!message) { return responseJSON({ ok: false, error: "Tin nhắn trống." }, 400); } // ============================== // GỌI GEMINI // ============================== const endpoint = "https://generativelanguage.googleapis.com/" + "v1beta/models/" + MODEL + ":generateContent"; const geminiResponse = await fetch( endpoint, { method: "POST", headers: { "Content-Type": "application/json", "x-goog-api-key": env.GEMINI_API_KEY }, body: JSON.stringify({ contents: [ { role: "user", parts: [ { text: `Bạn là ChatLuk, một trợ lý AI Việt Nam. Hãy trả lời người dùng bằng tiếng Việt tự nhiên, thân thiện, dễ hiểu và hữu ích. Không tự nhận mình là ChatGPT. Không nói về API hoặc Cloudflare nếu người dùng không hỏi. Tin nhắn người dùng: ${message}` } ] } ] }) } ); // ============================== // ĐỌC GEMINI RESPONSE // ============================== const raw = await geminiResponse.text(); let data; try { data = JSON.parse(raw); } catch { console.error( "Gemini trả về dữ liệu không hợp lệ:", raw ); return responseJSON({ ok: false, error: "Gemini không trả về JSON hợp lệ.", details: raw.substring(0, 1000) }, 502); } // ============================== // GEMINI ERROR // ============================== if (!geminiResponse.ok) { console.error( "Gemini API ERROR:", data ); return responseJSON({ ok: false, error: data?.error?.message || "Gemini API trả về lỗi.", status: geminiResponse.status }, geminiResponse.status); } // ============================== // LẤY CÂU TRẢ LỜI // ============================== const answer = data?.candidates?.[0]?.content?.parts ?.map(part => part.text || "") .join("") .trim(); if (!answer) { console.error( "Không tìm thấy câu trả lời:", data ); return responseJSON({ ok: false, error: "Gemini không trả về nội dung.", details: data }, 502); } // ============================== // TRẢ KẾT QUẢ CHO CHATLUK // ============================== return responseJSON({ ok: true, answer: answer }); } catch (error) { console.error( "CHATLUK WORKER ERROR:", error ); return responseJSON({ ok: false, error: error?.message || "Worker xảy ra lỗi không xác định." }, 500); } } // ============================== // URL KHÔNG TỒN TẠI // ============================== return responseJSON({ ok: false, error: "ChatLuk API không tồn tại." }, 404); } };
